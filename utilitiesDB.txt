-- Property Management System Database Schema
-- Updated and Optimized Version

-- DROP SCHEMA IF EXISTS
DROP SCHEMA IF EXISTS public CASCADE;
CREATE SCHEMA public AUTHORIZATION pg_database_owner;
COMMENT ON SCHEMA public IS 'Property Management System - Main Schema';

-- ==============================================
-- BASE LOOKUP TABLES
-- ==============================================

-- Amenities
DROP TABLE IF EXISTS amenities CASCADE;
CREATE TABLE amenities (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    description TEXT,
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
COMMENT ON TABLE amenities IS 'Property amenities catalog';

-- Utilities
DROP TABLE IF EXISTS utilities CASCADE;
CREATE TABLE utilities (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    description TEXT,
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
COMMENT ON TABLE utilities IS 'Utility types catalog';

-- Blacklist Categories
DROP TABLE IF EXISTS blacklist_categories CASCADE;
CREATE TABLE blacklist_categories (
    id SERIAL PRIMARY KEY,
    category_name VARCHAR(100) NOT NULL UNIQUE,
    description TEXT,
    default_severity VARCHAR(20) DEFAULT 'medium' CHECK (default_severity IN ('low', 'medium', 'high', 'severe')),
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
COMMENT ON TABLE blacklist_categories IS 'Categories for tenant blacklisting';

-- Document Categories
DROP TABLE IF EXISTS document_categories CASCADE;
CREATE TABLE document_categories (
    id SERIAL PRIMARY KEY,
    category_name VARCHAR(100) NOT NULL UNIQUE,
    description TEXT,
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
COMMENT ON TABLE document_categories IS 'Document classification categories';

-- Expense Categories
DROP TABLE IF EXISTS expense_categories CASCADE;
CREATE TABLE expense_categories (
    id SERIAL PRIMARY KEY,
    category_name VARCHAR(100) NOT NULL UNIQUE,
    description TEXT,
    parent_category_id INTEGER REFERENCES expense_categories(id) ON DELETE SET NULL,
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
COMMENT ON TABLE expense_categories IS 'Hierarchical expense categories';

-- User Roles
DROP TABLE IF EXISTS user_roles CASCADE;
CREATE TABLE user_roles (
    id SERIAL PRIMARY KEY,
    role_name VARCHAR(50) NOT NULL UNIQUE,
    description TEXT,
    permissions JSONB DEFAULT '{}'::jsonb,
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
COMMENT ON TABLE user_roles IS 'User role definitions with JSONB permissions';

-- ==============================================
-- PROPERTIES AND UNITS
-- ==============================================

-- Properties
DROP TABLE IF EXISTS properties CASCADE;
CREATE TABLE properties (
    id SERIAL PRIMARY KEY,
    property_name VARCHAR(255) NOT NULL,
    address TEXT NOT NULL,
    property_type VARCHAR(50) NOT NULL CHECK (property_type IN ('Apartment', 'House', 'Condominium', 'Townhouse', 'Studio', 'Commercial', 'Mixed-Use', 'Other')),
    total_units INTEGER DEFAULT 1 CHECK (total_units > 0),
    size_sq_ft INTEGER CHECK (size_sq_ft > 0),
    year_built INTEGER CHECK (year_built > 1800 AND year_built <= EXTRACT(YEAR FROM CURRENT_DATE)),
    monthly_rent NUMERIC(10, 2) CHECK (monthly_rent >= 0),
    security_deposit NUMERIC(10, 2) CHECK (security_deposit >= 0),
    description TEXT,
    latitude NUMERIC(10, 8),
    longitude NUMERIC(11, 8),
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT unique_property_address UNIQUE(address)
);
COMMENT ON TABLE properties IS 'Property master records';

-- Units
DROP TABLE IF EXISTS units CASCADE;
CREATE TABLE units (
    id SERIAL PRIMARY KEY,
    property_id INTEGER NOT NULL REFERENCES properties(id) ON DELETE CASCADE,
    unit_number VARCHAR(50) NOT NULL,
    bedrooms INTEGER DEFAULT 0 CHECK (bedrooms >= 0),
    bathrooms NUMERIC(3, 1) DEFAULT 0 CHECK (bathrooms >= 0),
    size_sq_ft INTEGER CHECK (size_sq_ft > 0),
    floor_number INTEGER,
    monthly_rent NUMERIC(10, 2) CHECK (monthly_rent >= 0),
    security_deposit NUMERIC(10, 2) CHECK (security_deposit >= 0),
    occupancy_status VARCHAR(20) DEFAULT 'vacant' CHECK (occupancy_status IN ('occupied', 'vacant', 'maintenance', 'reserved')),
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT unique_unit_per_property UNIQUE(property_id, unit_number)
);
COMMENT ON TABLE units IS 'Individual rental units within properties';

-- ==============================================
-- TENANTS
-- ==============================================

DROP TABLE IF EXISTS tenants CASCADE;
CREATE TABLE tenants (
    id SERIAL PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE,
    phone VARCHAR(20),
    alternate_phone VARCHAR(20),
    date_of_birth DATE CHECK (date_of_birth < CURRENT_DATE),
    identification_type VARCHAR(50) CHECK (identification_type IN ('SSN', 'Driver License', 'Passport', 'National ID', 'Other')),
    identification_number VARCHAR(100),
    emergency_contact_name VARCHAR(200),
    emergency_contact_phone VARCHAR(20),
    emergency_contact_relationship VARCHAR(100),
    employment_status VARCHAR(50) CHECK (employment_status IN ('Employed', 'Self-Employed', 'Unemployed', 'Student', 'Retired')),
    employer_name VARCHAR(255),
    employer_phone VARCHAR(20),
    monthly_income NUMERIC(10, 2) CHECK (monthly_income >= 0),
    previous_address TEXT,
    credit_score INTEGER CHECK (credit_score BETWEEN 300 AND 850),
    
    -- Blacklist fields
    is_blacklisted BOOLEAN DEFAULT false,
    blacklist_category_id INTEGER REFERENCES blacklist_categories(id) ON DELETE SET NULL,
    blacklist_reason TEXT,
    blacklisted_date TIMESTAMP,
    blacklisted_by INTEGER, -- Will reference users(id) after users table is created
    blacklist_notes TEXT,
    blacklist_severity VARCHAR(20) CHECK (blacklist_severity IN ('low', 'medium', 'high', 'severe')),
    blacklist_expires_at DATE,
    
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    CONSTRAINT check_blacklist_fields CHECK (
        (is_blacklisted = false) OR 
        (is_blacklisted = true AND blacklist_reason IS NOT NULL AND blacklisted_date IS NOT NULL)
    )
);
COMMENT ON TABLE tenants IS 'Tenant personal information and blacklist status';

-- ==============================================
-- USERS
-- ==============================================

DROP TABLE IF EXISTS users CASCADE;
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(100) NOT NULL UNIQUE,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    phone VARCHAR(20),
    role_id INTEGER NOT NULL REFERENCES user_roles(id) ON DELETE RESTRICT,
    tenant_id INTEGER REFERENCES tenants(id) ON DELETE SET NULL,
    
    -- Account status
    is_active BOOLEAN DEFAULT true,
    is_verified BOOLEAN DEFAULT false,
    is_locked BOOLEAN DEFAULT false,
    
    -- Authentication
    last_login TIMESTAMP,
    password_reset_token VARCHAR(255),
    password_reset_expires TIMESTAMP,
    email_verification_token VARCHAR(255),
    email_verification_expires TIMESTAMP,
    two_factor_enabled BOOLEAN DEFAULT false,
    two_factor_secret VARCHAR(255),
    failed_login_attempts INTEGER DEFAULT 0,
    locked_until TIMESTAMP,
    
    -- Profile
    profile_image_url VARCHAR(500),
    timezone VARCHAR(50) DEFAULT 'Africa/Nairobi',
    language VARCHAR(10) DEFAULT 'en',
    notification_preferences JSONB DEFAULT '{"sms": false, "push": true, "email": true}'::jsonb,
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
COMMENT ON TABLE users IS 'System users with authentication and authorization';

-- Add foreign key for blacklisted_by after users table exists
ALTER TABLE tenants ADD CONSTRAINT fk_tenants_blacklisted_by 
    FOREIGN KEY (blacklisted_by) REFERENCES users(id) ON DELETE SET NULL;
-- ==============================================
-- LEASES
-- ==============================================

DROP TABLE IF EXISTS leases CASCADE;
CREATE TABLE leases (
    id SERIAL PRIMARY KEY,
    unit_id INTEGER NOT NULL REFERENCES units(id) ON DELETE RESTRICT,
    lease_number VARCHAR(100) NOT NULL UNIQUE,
    lease_type VARCHAR(50) DEFAULT 'Fixed Term' CHECK (lease_type IN ('Fixed Term', 'Month-to-Month', 'Week-to-Week', 'Corporate')),
    lease_status VARCHAR(50) DEFAULT 'draft' CHECK (lease_status IN ('draft', 'active', 'expired', 'terminated', 'renewed', 'pending_renewal')),
    
    -- Dates
    start_date DATE NOT NULL,
    end_date DATE,
    signed_date DATE,
    move_in_date DATE,
    move_out_date DATE,
    notice_date DATE,
    
    -- Financial terms
    monthly_rent NUMERIC(10, 2) NOT NULL CHECK (monthly_rent > 0),
    security_deposit NUMERIC(10, 2) NOT NULL CHECK (security_deposit >= 0),
    pet_deposit NUMERIC(10, 2) DEFAULT 0 CHECK (pet_deposit >= 0),
    late_fee NUMERIC(10, 2) DEFAULT 0 CHECK (late_fee >= 0),
    grace_period_days INTEGER DEFAULT 5 CHECK (grace_period_days >= 0),
    rent_due_day INTEGER DEFAULT 1 CHECK (rent_due_day BETWEEN 1 AND 31),
    
    -- Terms
    lease_terms TEXT,
    special_conditions TEXT,
    pets_allowed BOOLEAN DEFAULT false,
    smoking_allowed BOOLEAN DEFAULT false,
    subletting_allowed BOOLEAN DEFAULT false,
    
    -- Renewal tracking
    renewal_notice_days INTEGER DEFAULT 60,
    auto_renew BOOLEAN DEFAULT false,
    renewed_to_lease_id INTEGER REFERENCES leases(id) ON DELETE SET NULL,
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    CONSTRAINT check_lease_dates CHECK (end_date IS NULL OR end_date > start_date),
    CONSTRAINT check_move_in_date CHECK (move_in_date IS NULL OR move_in_date >= start_date),
    CONSTRAINT check_move_out_date CHECK (move_out_date IS NULL OR (move_in_date IS NULL OR move_out_date >= move_in_date)),
    CONSTRAINT check_signed_date CHECK (signed_date IS NULL OR signed_date <= start_date)
);
COMMENT ON TABLE leases IS 'Lease agreements between landlord and tenants';

-- ==============================================
-- JUNCTION TABLES
-- ==============================================

-- Lease Tenants (Many-to-Many)
DROP TABLE IF EXISTS lease_tenants CASCADE;
CREATE TABLE lease_tenants (
    id SERIAL PRIMARY KEY,
    lease_id INTEGER NOT NULL REFERENCES leases(id) ON DELETE CASCADE,
    tenant_id INTEGER NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    is_primary_tenant BOOLEAN DEFAULT false,
    tenant_type VARCHAR(50) DEFAULT 'Tenant' CHECK (tenant_type IN ('Tenant', 'Co-Tenant', 'Guarantor', 'Occupant', 'Emergency Contact')),
    responsibility_percentage NUMERIC(5, 2) DEFAULT 100.00 CHECK (responsibility_percentage BETWEEN 0 AND 100),
    added_date DATE DEFAULT CURRENT_DATE,
    removed_date DATE,
    removal_reason TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT unique_lease_tenant UNIQUE(lease_id, tenant_id),
    CONSTRAINT check_removed_date CHECK (removed_date IS NULL OR removed_date >= added_date)
);
COMMENT ON TABLE lease_tenants IS 'Association between leases and tenants';

-- Property Amenities
DROP TABLE IF EXISTS property_amenities CASCADE;
CREATE TABLE property_amenities (
    id SERIAL PRIMARY KEY,
    property_id INTEGER NOT NULL REFERENCES properties(id) ON DELETE CASCADE,
    amenity_id INTEGER NOT NULL REFERENCES amenities(id) ON DELETE CASCADE,
    quantity INTEGER DEFAULT 1 CHECK (quantity > 0),
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT unique_property_amenity UNIQUE(property_id, amenity_id)
);

-- Property Utilities
DROP TABLE IF EXISTS property_utilities CASCADE;
CREATE TABLE property_utilities (
    id SERIAL PRIMARY KEY,
    property_id INTEGER NOT NULL REFERENCES properties(id) ON DELETE CASCADE,
    utility_id INTEGER NOT NULL REFERENCES utilities(id) ON DELETE CASCADE,
    included BOOLEAN DEFAULT false,
    provider_name VARCHAR(255),
    account_number VARCHAR(100),
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT unique_property_utility UNIQUE(property_id, utility_id)
);

-- Unit Utilities
DROP TABLE IF EXISTS unit_utilities CASCADE;
CREATE TABLE unit_utilities (
    id SERIAL PRIMARY KEY,
    unit_id INTEGER NOT NULL REFERENCES units(id) ON DELETE CASCADE,
    utility_id INTEGER NOT NULL REFERENCES utilities(id) ON DELETE CASCADE,
    included BOOLEAN DEFAULT false,
    meter_number VARCHAR(100),
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT unique_unit_utility UNIQUE(unit_id, utility_id)
);

-- ==============================================
-- FINANCIAL TABLES
-- ==============================================

-- Rent Payments
DROP TABLE IF EXISTS rent_payments CASCADE;
CREATE TABLE rent_payments (
    id SERIAL PRIMARY KEY,
    lease_id INTEGER NOT NULL REFERENCES leases(id) ON DELETE RESTRICT,
    payment_date DATE,
    due_date DATE NOT NULL,
    amount_due NUMERIC(10, 2) NOT NULL CHECK (amount_due >= 0),
    amount_paid NUMERIC(10, 2) DEFAULT 0 CHECK (amount_paid >= 0),
    payment_method VARCHAR(50) CHECK (payment_method IN ('Cash', 'Check', 'Bank Transfer', 'Credit Card', 'Debit Card', 'Online Payment', 'Money Order', 'M-Pesa', 'Airtel Money', 'Other')),
    payment_reference VARCHAR(255),
    transaction_id VARCHAR(255),
    late_fee NUMERIC(10, 2) DEFAULT 0 CHECK (late_fee >= 0),
    payment_status VARCHAR(50) DEFAULT 'pending' CHECK (payment_status IN ('pending', 'paid', 'partial', 'overdue', 'failed', 'refunded')),
    notes TEXT,
    processed_by varchar(255),
    utilities_charges NUMERIC(10, 2) DEFAULT 0 CHECK (utilities_charges >= 0),
    processed_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT check_payment_date CHECK (
        (payment_status IN ('paid', 'partial', 'refunded') AND payment_date IS NOT NULL) OR 
        payment_status IN ('pending', 'overdue', 'failed')
    ),
    CONSTRAINT check_amount_paid 
CHECK (amount_paid <= amount_due + COALESCE(utilities_charges, 0) + COALESCE(late_fee, 0));
);
COMMENT ON TABLE rent_payments IS 'Rent payment records and status tracking';

-- Payment Submissions (Tenant-initiated payment proof)
DROP TABLE IF EXISTS payment_submissions CASCADE;
CREATE TABLE payment_submissions (
    id SERIAL PRIMARY KEY,
    tenant_id INTEGER NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    lease_id INTEGER NOT NULL REFERENCES leases(id) ON DELETE CASCADE,
    amount NUMERIC(10, 2) NOT NULL CHECK (amount > 0),
    payment_method VARCHAR(50) NOT NULL CHECK (payment_method IN ('bank_transfer', 'mpesa', 'airtel_money', 'cash', 'check', 'other')),
    transaction_reference VARCHAR(255) NOT NULL,
    transaction_date DATE NOT NULL CHECK (transaction_date <= CURRENT_DATE),
    
    -- Submission tracking
    submission_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    submitted_by INTEGER REFERENCES users(id) ON DELETE SET NULL,
    
    -- Verification tracking
    verification_status VARCHAR(50) DEFAULT 'pending' CHECK (verification_status IN ('pending', 'verified', 'rejected', 'under_review')),
    verified_date TIMESTAMP,
    verified_by INTEGER REFERENCES users(id) ON DELETE SET NULL,
    rejection_reason TEXT,
    
    -- Notes
    notes TEXT,
    admin_notes TEXT,
    
    -- Attachments
    attachment_urls TEXT[],
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    CONSTRAINT check_verification_fields CHECK (
        (verification_status = 'pending' AND verified_date IS NULL AND verified_by IS NULL) OR
        (verification_status IN ('verified', 'rejected') AND verified_date IS NOT NULL AND verified_by IS NOT NULL)
    ),
    CONSTRAINT check_rejection_reason CHECK (
        (verification_status = 'rejected' AND rejection_reason IS NOT NULL) OR
        (verification_status != 'rejected')
    )
);
COMMENT ON TABLE payment_submissions IS 'Tenant-submitted payment proofs awaiting verification';

-- Security Deposits
DROP TABLE IF EXISTS security_deposits CASCADE;
CREATE TABLE security_deposits (
    id SERIAL PRIMARY KEY,
    lease_id INTEGER NOT NULL REFERENCES leases(id) ON DELETE RESTRICT,
    deposit_type VARCHAR(50) DEFAULT 'Security' CHECK (deposit_type IN ('Security', 'Pet', 'Key', 'Cleaning', 'Damage', 'Other')),
    amount_collected NUMERIC(10, 2) NOT NULL CHECK (amount_collected > 0),
    collection_date DATE NOT NULL,
    collection_method VARCHAR(50),
    collection_reference VARCHAR(255),
    
    -- Return tracking
    amount_returned NUMERIC(10, 2) DEFAULT 0 CHECK (amount_returned >= 0),
    return_date DATE,
    return_method VARCHAR(50),
    return_reference VARCHAR(255),
    
    -- Deductions
    deductions NUMERIC(10, 2) DEFAULT 0 CHECK (deductions >= 0),
    deduction_reason TEXT,
    deduction_itemization JSONB,
    
    status VARCHAR(50) DEFAULT 'held' CHECK (status IN ('held', 'partially_returned', 'fully_returned', 'forfeited')),
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    CONSTRAINT check_amounts CHECK (amount_returned + deductions <= amount_collected),
    CONSTRAINT check_return_date CHECK (return_date IS NULL OR return_date >= collection_date),
    CONSTRAINT check_return_fields CHECK (
        (status IN ('partially_returned', 'fully_returned') AND return_date IS NOT NULL) OR
        (status IN ('held', 'forfeited'))
    )
);
COMMENT ON TABLE security_deposits IS 'Security deposit tracking with deductions and returns';

-- Property Expenses
DROP TABLE IF EXISTS property_expenses CASCADE;
CREATE TABLE property_expenses (
    id SERIAL PRIMARY KEY,
    property_id INTEGER NOT NULL REFERENCES properties(id) ON DELETE CASCADE,
    category_id INTEGER REFERENCES expense_categories(id) ON DELETE SET NULL,
    expense_type VARCHAR(100) NOT NULL,
    vendor_name VARCHAR(255),
    amount NUMERIC(10, 2) NOT NULL CHECK (amount > 0),
    frequency VARCHAR(20) DEFAULT 'one-time' CHECK (frequency IN ('monthly', 'quarterly', 'semi-annual', 'annual', 'one-time')),
    
    -- Date tracking
    expense_date DATE DEFAULT CURRENT_DATE,
    start_date DATE DEFAULT CURRENT_DATE,
    end_date DATE,
    next_due_date DATE,
    
    -- Status
    is_active BOOLEAN DEFAULT true,
    is_paid BOOLEAN DEFAULT false,
    
    -- Details
    description TEXT,
    invoice_number VARCHAR(100),
    receipt_url VARCHAR(500),
    
    -- Tracking
    created_by VARCHAR(250),
  
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    CONSTRAINT check_dates CHECK (end_date IS NULL OR end_date >= start_date),
    CONSTRAINT check_paid_fields CHECK (
        (is_paid = true AND paid_date IS NOT NULL) OR
        (is_paid = false)
    )
);
COMMENT ON TABLE property_expenses IS 'Property-related expenses and recurring costs';

-- Expense Payments (Track individual payments for recurring expenses)
DROP TABLE IF EXISTS expense_payments CASCADE;
CREATE TABLE expense_payments (
    id SERIAL PRIMARY KEY,
    expense_id INTEGER NOT NULL REFERENCES property_expenses(id) ON DELETE CASCADE,
    payment_date DATE NOT NULL,
    amount_paid NUMERIC(10, 2) NOT NULL CHECK (amount_paid > 0),
    payment_method VARCHAR(50),
    payment_reference VARCHAR(255),
    paid_by INTEGER REFERENCES users(id) ON DELETE SET NULL,
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
COMMENT ON TABLE expense_payments IS 'Individual payments for property expenses';

DROP TABLE IF EXISTS maintenance_requests CASCADE;
CREATE TABLE maintenance_requests (
    id SERIAL PRIMARY KEY,
    unit_id INTEGER NOT NULL REFERENCES units(id) ON DELETE RESTRICT,
    tenant_id INTEGER REFERENCES tenants(id) ON DELETE SET NULL,
    lease_id INTEGER REFERENCES leases(id) ON DELETE SET NULL,
    
    -- Request details
    request_title VARCHAR(255) NOT NULL,
    description TEXT NOT NULL,
    priority VARCHAR(20) DEFAULT 'medium' CHECK (priority IN ('low', 'medium', 'high', 'emergency')),
    category VARCHAR(100) CHECK (category IN ('Plumbing', 'Electrical', 'HVAC', 'Appliances', 'Structural', 'Pest Control', 'Cleaning', 'Security', 'Painting', 'Flooring', 'Windows/Doors', 'Landscaping', 'Other')),
    
    -- Status tracking
    status VARCHAR(50) DEFAULT 'open' CHECK (status IN ('open', 'assigned', 'in_progress', 'completed', 'cancelled', 'on_hold', 'pending_parts')),
    
    -- Date tracking
    requested_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    acknowledged_date TIMESTAMP,
    scheduled_date TIMESTAMP,
    started_date TIMESTAMP,
    completed_date TIMESTAMP,
    
    -- Assignment
    assigned_to INTEGER REFERENCES users(id) ON DELETE SET NULL,
    assigned_date TIMESTAMP,
    vendor_name VARCHAR(255),
    vendor_contact VARCHAR(100),
    
    -- Financial
    estimated_cost NUMERIC(10, 2) CHECK (estimated_cost >= 0),
    actual_cost NUMERIC(10, 2) CHECK (actual_cost >= 0),
    invoice_number VARCHAR(100),
    
    -- Access
    entry_allowed BOOLEAN DEFAULT false,
    access_instructions TEXT,
    
    -- Notes
    tenant_notes TEXT,
    management_notes TEXT,
    completion_notes TEXT,
    
    -- Attachments
    attachment_urls TEXT[],
    
    -- Ratings
    tenant_rating INTEGER CHECK (tenant_rating BETWEEN 1 AND 5),
    tenant_feedback TEXT,
    
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    CONSTRAINT check_completed_date CHECK (
        completed_date IS NULL OR 
        DATE(completed_date) >= DATE(requested_date)
    )
);
COMMENT ON TABLE maintenance_requests IS 'Maintenance and repair requests';

-- ==============================================
-- COMMUNICATIONS
-- ==============================================

DROP TABLE IF EXISTS communications CASCADE;
CREATE TABLE communications (
    id SERIAL PRIMARY KEY,
    
    -- Participants
    tenant_id INTEGER REFERENCES tenants(id) ON DELETE CASCADE,
    lease_id INTEGER REFERENCES leases(id) ON DELETE CASCADE,
    sender_id INTEGER REFERENCES users(id) ON DELETE SET NULL,
    recipient_id INTEGER REFERENCES users(id) ON DELETE SET NULL,
    
    -- Communication details
    communication_type VARCHAR(50) DEFAULT 'Email' CHECK (communication_type IN ('Email', 'Phone Call', 'Text Message', 'In-Person', 'Letter', 'Notice', 'Portal Message', 'WhatsApp', 'Other')),
    direction VARCHAR(20) CHECK (direction IN ('inbound', 'outbound')),
    subject VARCHAR(255),
    message_content TEXT,
    
    -- Metadata
    communication_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    staff_member VARCHAR(255),
    
    -- Follow-up
    follow_up_required BOOLEAN DEFAULT false,
    follow_up_date DATE,
    follow_up_completed BOOLEAN DEFAULT false,
    
    -- Attachments
    attachment_urls TEXT[],
    
    -- Tracking
    is_read BOOLEAN DEFAULT false,
    read_at TIMESTAMP,
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    CONSTRAINT check_follow_up_date CHECK (
        (follow_up_required = false) OR
        (follow_up_required = true AND follow_up_date IS NOT NULL)
    )
);
COMMENT ON TABLE communications IS 'All communications with tenants';

CREATE INDEX idx_communications_tenant ON communications(tenant_id);
CREATE INDEX idx_communications_lease ON communications(lease_id);
CREATE INDEX idx_communications_date ON communications(communication_date);
CREATE INDEX idx_communications_sender ON communications(sender_id);
CREATE INDEX idx_communications_recipient ON communications(recipient_id);
CREATE INDEX idx_communications_follow_up ON communications(follow_up_required, follow_up_date) WHERE follow_up_required = true;

-- ==============================================
-- DOCUMENT MANAGEMENT
-- ==============================================

DROP TABLE IF EXISTS documents CASCADE;
CREATE TABLE documents (
    id SERIAL PRIMARY KEY,
    document_name VARCHAR(255) NOT NULL,
    original_filename VARCHAR(255) NOT NULL,
    file_path VARCHAR(500) NOT NULL,

-- Add check constraint using a trigger instead of inline CHECK
CREATE OR REPLACE FUNCTION validate_document_associations()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.property_id IS NULL AND 
       NEW.unit_id IS NULL AND 
       NEW.tenant_id IS NULL AND 
       NEW.lease_id IS NULL AND
       NEW.maintenance_request_id IS NULL THEN
        RAISE EXCEPTION 'Document must be associated with at least one entity (property, unit, tenant, lease, or maintenance request)';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_validate_document_associations
    BEFORE INSERT OR UPDATE ON documents
    FOR EACH ROW EXECUTE FUNCTION validate_document_associations();

-- Update timestamp trigger function
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Function to bill utilities and link to rent payments
CREATE OR REPLACE FUNCTION bill_utilities_to_rent(
    target_month INTEGER,
    target_year INTEGER
)
RETURNS TABLE(
    rent_payment_id INTEGER,
    lease_id INTEGER,
    lease_number VARCHAR(100),
    utilities_total NUMERIC(10,2),
    updated_count INTEGER
)
LANGUAGE plpgsql
AS $$
DECLARE
    billing_month_date DATE;
    affected_rows INTEGER := 0;
    utility_record RECORD;
    rent_due_date DATE;
    rent_payment_record_id INTEGER;
BEGIN
    -- Calculate the billing month date
    billing_month_date := DATE(target_year || '-' || LPAD(target_month::TEXT, 2, '0') || '-01');
    
    -- Loop through all pending utility charges for the target month
    FOR utility_record IN
        SELECT 
            uc.id as utility_charge_id,
            uc.lease_id as uc_lease_id,
            uc.total_utility_charges,
            l.lease_number,
            l.rent_due_day
        FROM utility_charges uc
        JOIN leases l ON uc.lease_id = l.id
        WHERE uc.billing_month = billing_month_date
        AND uc.charge_status = 'pending'
        AND l.lease_status = 'active'
    LOOP
        -- Calculate due date
        rent_due_date := billing_month_date + (COALESCE(utility_record.rent_due_day, 1) - 1) * INTERVAL '1 day';
        
        -- Ensure due date doesn't exceed month boundaries
        IF rent_due_date >= (billing_month_date + INTERVAL '1 month') THEN
            rent_due_date := (billing_month_date + INTERVAL '1 month' - INTERVAL '1 day')::DATE;
        END IF;
        
        -- Find the rent payment record
        SELECT rp.id INTO rent_payment_record_id
        FROM rent_payments rp
        WHERE rp.lease_id = utility_record.uc_lease_id
        AND rp.due_date = rent_due_date;
        
        -- If rent payment exists, update it
        IF rent_payment_record_id IS NOT NULL THEN
            -- Update the rent payment with utility charges
            UPDATE rent_payments
            SET 
                utilities_charges = utility_record.total_utility_charges,
                updated_at = CURRENT_TIMESTAMP
            WHERE id = rent_payment_record_id;
            
            -- Update utility charge status to 'billed'
            UPDATE utility_charges
            SET 
                charge_status = 'billed',
                bill_date = CURRENT_DATE,
                due_date = rent_due_date,
                updated_at = CURRENT_TIMESTAMP
            WHERE id = utility_record.utility_charge_id;
            
            affected_rows := affected_rows + 1;
            
            -- Return the result for this record
            RETURN QUERY SELECT 
                rent_payment_record_id,
                utility_record.uc_lease_id,
                utility_record.lease_number,
                utility_record.total_utility_charges,
                1;
        ELSE
            -- Log that rent payment doesn't exist
            RAISE NOTICE 'No rent payment found for lease % on date %', 
                utility_record.lease_number, rent_due_date;
        END IF;
    END LOOP;
    
    -- Log the execution
    INSERT INTO cron_job_log (
        job_name,
        records_affected,
        success,
        job_parameters
    ) VALUES (
        'bill_utilities_to_rent',
        affected_rows,
        true,
        jsonb_build_object(
            'month', target_month,
            'year', target_year
        )
    );
    
    RETURN;
    
EXCEPTION WHEN OTHERS THEN
    -- Log errors
    INSERT INTO cron_job_log (
        job_name,
        success,
        error_message,
        job_parameters
    ) VALUES (
        'bill_utilities_to_rent',
        false,
        SQLERRM,
        jsonb_build_object(
            'month', target_month,
            'year', target_year
        )
    );
    RAISE;
END;
$$;

COMMENT ON FUNCTION bill_utilities_to_rent(INTEGER, INTEGER) IS 'Bills pending utilities to rent payments and marks them as billed';

-- Generate lease number
CREATE OR REPLACE FUNCTION generate_lease_number()
RETURNS TRIGGER AS $$
DECLARE
    property_code VARCHAR(10);
    unit_code VARCHAR(10);
    year_code VARCHAR(4);
    sequence_num INTEGER;
    new_lease_number VARCHAR(100);
BEGIN
    -- Get property and unit info
    SELECT 
        UPPER(LEFT(REGEXP_REPLACE(p.property_name, '[^a-zA-Z0-9]', '', 'g'), 3)),
        COALESCE(UPPER(REGEXP_REPLACE(u.unit_number, '[^a-zA-Z0-9]', '', 'g')), 'MAIN'),
        EXTRACT(YEAR FROM CURRENT_DATE)::VARCHAR
    INTO property_code, unit_code, year_code
    FROM units u
    JOIN properties p ON u.property_id = p.id
    WHERE u.id = NEW.unit_id;
    
    -- Ensure we have at least some property code
    IF property_code IS NULL OR property_code = '' THEN
        property_code := 'UNK';
    END IF;
    
    -- Get next sequence number for this property/year
    SELECT COALESCE(MAX(
        CASE 
            WHEN lease_number ~ '.*-[0-9]+$' 
            THEN CAST(SUBSTRING(lease_number FROM '([0-9]+)$') AS INTEGER)
            ELSE 0
        END
    ), 0) + 1
    INTO sequence_num
    FROM leases l
    JOIN units u ON l.unit_id = u.id
    JOIN properties p ON u.property_id = p.id
    WHERE l.lease_number LIKE property_code || '%' || year_code || '%';
    
    -- Generate lease number: PROPERTYCODE-UNITCODE-YEAR-SEQUENCE
    new_lease_number := property_code || '-' || unit_code || '-' || year_code || '-' || LPAD(sequence_num::VARCHAR, 3, '0');
    
    NEW.lease_number := new_lease_number;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

COMMENT ON FUNCTION generate_lease_number() IS 'Auto-generates unique lease numbers';

-- Update unit occupancy based on lease status
CREATE OR REPLACE FUNCTION update_unit_occupancy()
RETURNS TRIGGER AS $$
BEGIN
    -- When lease becomes active, mark unit as occupied
    IF NEW.lease_status = 'active' AND (OLD.lease_status IS NULL OR OLD.lease_status != 'active') THEN
        UPDATE units SET occupancy_status = 'occupied' WHERE id = NEW.unit_id;
    END IF;
    
    -- When lease ends/terminates, mark unit as vacant (if no other active leases)
    IF NEW.lease_status IN ('expired', 'terminated') AND OLD.lease_status = 'active' THEN
        IF NOT EXISTS (
            SELECT 1 FROM leases 
            WHERE unit_id = NEW.unit_id 
            AND lease_status = 'active' 
            AND id != NEW.id
        ) THEN
            UPDATE units SET occupancy_status = 'vacant' WHERE id = NEW.unit_id;
        END IF;
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

COMMENT ON FUNCTION update_unit_occupancy() IS 'Updates unit occupancy status based on lease status';

-- Validate user-tenant relationship
CREATE OR REPLACE FUNCTION validate_user_tenant_relationship()
RETURNS TRIGGER AS $$
DECLARE
    tenant_role_id INTEGER;
    role_name_check VARCHAR(50);
BEGIN
    -- Get the role name for the user's role
    SELECT ur.role_name, ur.id 
    INTO role_name_check, tenant_role_id
    FROM user_roles ur 
    WHERE ur.id = NEW.role_id;
    
    -- If role is 'Tenant', tenant_id must not be NULL
    IF role_name_check = 'Tenant' THEN
        IF NEW.tenant_id IS NULL THEN
            RAISE EXCEPTION 'Users with Tenant role must have a tenant_id';
        END IF;
    END IF;
    
    -- If role is not 'Tenant', tenant_id must be NULL
    IF role_name_check IS DISTINCT FROM 'Tenant' OR role_name_check IS NULL THEN
        IF NEW.tenant_id IS NOT NULL THEN
            RAISE EXCEPTION 'Users with role "%" cannot have a tenant_id', role_name_check;
        END IF;
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

COMMENT ON FUNCTION validate_user_tenant_relationship() IS 'Ensures proper user-tenant associations based on role';


-- Process verified payment submissions
-- Process verified payment submissions
CREATE OR REPLACE FUNCTION process_verified_payment_submission()
RETURNS TRIGGER AS $$
DECLARE
    rent_payment_record RECORD;
    remaining_amount NUMERIC(10,2);
    amount_to_apply NUMERIC(10,2);
    mapped_payment_method VARCHAR(50);
    tenant_user_id INTEGER;
    property_name VARCHAR(255);
    unit_number VARCHAR(50);
    lease_number VARCHAR(100);
    total_applied NUMERIC(10,2) := 0;
    payments_updated INTEGER := 0;
BEGIN
    -- Only process when status changes to 'verified'
    IF NEW.verification_status = 'verified' AND 
       (OLD.verification_status IS NULL OR OLD.verification_status != 'verified') THEN
        
        -- Map payment methods
        mapped_payment_method := CASE NEW.payment_method
            WHEN 'mpesa' THEN 'M-Pesa'
            WHEN 'airtel_money' THEN 'Airtel Money'
            WHEN 'bank_transfer' THEN 'Bank Transfer'
            WHEN 'cash' THEN 'Cash'
            WHEN 'check' THEN 'Check'
            ELSE 'Other'
        END;
        
        -- Get property and lease info for notifications
        SELECT 
            p.property_name,
            u.unit_number,
            l.lease_number
        INTO
            property_name,
            unit_number,
            lease_number
        FROM leases l
        JOIN units u ON l.unit_id = u.id
        JOIN properties p ON u.property_id = p.id
        WHERE l.id = NEW.lease_id;
        
        -- Get tenant's user_id if they have an account
        SELECT us.id INTO tenant_user_id
        FROM users us
        WHERE us.tenant_id = NEW.tenant_id
        LIMIT 1;
        
        remaining_amount := NEW.amount;
        
        -- Loop through pending/overdue/partial payments for this lease, oldest first
        FOR rent_payment_record IN
            SELECT 
                id, 
                amount_due, 
                utilities_charges, 
                late_fee, 
                amount_paid,
                payment_status,
                due_date
            FROM rent_payments
            WHERE lease_id = NEW.lease_id
            AND payment_status IN ('pending', 'overdue', 'partial')
            ORDER BY due_date ASC
        LOOP
            -- Calculate total amount due (rent + utilities + late fee)
            DECLARE
                total_due NUMERIC(10,2);
                current_balance NUMERIC(10,2);
                new_amount_paid NUMERIC(10,2);
                new_status VARCHAR(50);
            BEGIN
                total_due := rent_payment_record.amount_due + 
                            COALESCE(rent_payment_record.utilities_charges, 0) + 
                            COALESCE(rent_payment_record.late_fee, 0);
                current_balance := total_due - rent_payment_record.amount_paid;
                
                -- Apply as much as possible to this payment
                IF current_balance > 0 AND remaining_amount > 0 THEN
                    amount_to_apply := LEAST(remaining_amount, current_balance);
                    new_amount_paid := rent_payment_record.amount_paid + amount_to_apply;
                    
                    -- Determine new status
                    IF new_amount_paid >= total_due THEN
                        new_status := 'paid';
                    ELSIF new_amount_paid > 0 THEN
                        new_status := 'partial';
                    ELSE
                        new_status := rent_payment_record.payment_status;
                    END IF;
                    
                    -- Update the rent payment
                    UPDATE rent_payments
                    SET 
                        amount_paid = new_amount_paid,
                        payment_status = new_status,
                        payment_method = mapped_payment_method,
                        payment_reference = NEW.transaction_reference,
                        payment_date = CASE 
                            WHEN new_status = 'paid' THEN NEW.transaction_date 
                            ELSE payment_date 
                        END,
                        processed_by = NEW.verified_by::VARCHAR,
                        processed_at = CURRENT_TIMESTAMP,
                        updated_at = CURRENT_TIMESTAMP
                    WHERE id = rent_payment_record.id;
                    
                    -- ✅ UPDATE UTILITY CHARGES STATUS TO PAID
                    IF rent_payment_record.utilities_charges > 0 AND new_status IN ('paid', 'partial') THEN
                        UPDATE utility_charges
                        SET 
                            charge_status = CASE 
                                WHEN new_status = 'paid' THEN 'paid'
                                ELSE 'billed'
                            END,
                            payment_date = CASE 
                                WHEN new_status = 'paid' THEN NEW.transaction_date
                                ELSE payment_date
                            END,
                            updated_at = CURRENT_TIMESTAMP
                        WHERE lease_id = NEW.lease_id
                        AND DATE_TRUNC('month', billing_month) = DATE_TRUNC('month', rent_payment_record.due_date)
                        AND charge_status = 'billed';
                    END IF;
                    
                    -- Log in payment history
                    INSERT INTO payment_history (
                        payment_id,
                        change_type,
                        old_status,
                        new_status,
                        old_amount,
                        new_amount,
                        changed_by,
                        additional_data
                    ) VALUES (
                        rent_payment_record.id,
                        'payment_received',
                        rent_payment_record.payment_status,
                        new_status,
                        rent_payment_record.amount_paid,
                        new_amount_paid,
                        NEW.verified_by,
                        jsonb_build_object(
                            'submission_id', NEW.id,
                            'amount_applied', amount_to_apply,
                            'remaining_balance', current_balance - amount_to_apply,
                            'included_utilities', rent_payment_record.utilities_charges,
                            'included_late_fee', rent_payment_record.late_fee
                        )
                    );
                    
                    total_applied := total_applied + amount_to_apply;
                    remaining_amount := remaining_amount - amount_to_apply;
                    payments_updated := payments_updated + 1;
                    
                    -- Stop if we've applied all the payment
                    EXIT WHEN remaining_amount <= 0;
                END IF;
            END;
        END LOOP;
        
        -- Create notification for tenant (if they have a user account)
        IF tenant_user_id IS NOT NULL THEN
            INSERT INTO user_notifications (
                user_id,
                notification_type,
                title,
                message,
                priority,
                is_urgent,
                related_resource_type,
                related_resource_id,
                delivery_method
            ) VALUES (
                tenant_user_id,
                'payment_verified',
                'Payment Verified',
                'Your payment of KES ' || NEW.amount || ' via ' || mapped_payment_method || 
                ' for ' || property_name || ' - Unit ' || unit_number || 
                ' has been verified and applied to ' || payments_updated || ' rent payment(s). ' ||
                CASE 
                    WHEN remaining_amount > 0 THEN 'Excess amount: KES ' || remaining_amount || ' will be applied to future payments.'
                    ELSE 'Payment fully applied.'
                END,
                'normal',
                false,
                'payment_submission',
                NEW.id,
                'all'
            );
        END IF;
        
        -- Create notification for admins/managers
        INSERT INTO user_notifications (
            user_id,
            notification_type,
            title,
            message,
            priority,
            related_resource_type,
            related_resource_id,
            target_scope,
            target_role_ids,
            delivery_method
        ) 
        SELECT 
            NULL,
            'payment_processed',
            'Payment Processed',
            'Payment of KES ' || NEW.amount || ' for lease ' || lease_number || 
            ' (' || property_name || ' - Unit ' || unit_number || ') has been verified. ' ||
            'Applied KES ' || total_applied || ' to ' || payments_updated || ' rent payment(s).',
            'low',
            'payment_submission',
            NEW.id,
            'role_based',
            ARRAY[1, 2], -- Admin and Manager role IDs
            'in_app';
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

COMMENT ON FUNCTION process_verified_payment_submission() IS 'Applies verified payment submissions to rent payments including utilities and creates notifications';

-- Trigger to update utility charges when rent payment is marked as paid
CREATE OR REPLACE FUNCTION update_utility_charges_on_payment()
RETURNS TRIGGER AS $$
BEGIN
    -- When payment status changes to 'paid' and utilities were included
    IF NEW.payment_status = 'paid' AND 
       OLD.payment_status != 'paid' AND 
       NEW.utilities_charges > 0 THEN
        
        UPDATE utility_charges
        SET 
            charge_status = 'paid',
            payment_date = NEW.payment_date,
            updated_at = CURRENT_TIMESTAMP
        WHERE lease_id = NEW.lease_id
        AND DATE_TRUNC('month', billing_month) = DATE_TRUNC('month', NEW.due_date)
        AND charge_status IN ('billed', 'pending');
        
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Create the trigger
DROP TRIGGER IF EXISTS trigger_update_utility_charges_on_payment ON rent_payments;
CREATE TRIGGER trigger_update_utility_charges_on_payment
    AFTER UPDATE ON rent_payments
    FOR EACH ROW
    WHEN (NEW.payment_status = 'paid' AND OLD.payment_status IS DISTINCT FROM 'paid')
    EXECUTE FUNCTION update_utility_charges_on_payment();

-- Generate monthly rent payments
CREATE OR REPLACE FUNCTION generate_monthly_rent_payments()
RETURNS void
LANGUAGE plpgsql
AS $$
DECLARE
    lease_record RECORD;
    current_due_date DATE;
    current_month_start DATE;
    affected_rows INTEGER := 0;
    system_user_id INTEGER;
    start_time TIMESTAMP;
BEGIN
    start_time := CURRENT_TIMESTAMP;
    
    -- Get the first day of the current month
    current_month_start := DATE_TRUNC('month', CURRENT_DATE);
    
    -- Try to get a system user ID for logging
    SELECT id INTO system_user_id 
    FROM users 
    WHERE username = 'system' 
    LIMIT 1;
    
    -- Loop through all active leases
    FOR lease_record IN 
        SELECT 
            l.id,
            l.unit_id,
            l.monthly_rent,
            l.rent_due_day,
            l.start_date,
            l.end_date,
            l.lease_status,
            l.late_fee
        FROM leases l
        WHERE l.lease_status = 'active'
        AND l.start_date <= current_month_start + INTERVAL '1 month' - INTERVAL '1 day'
        AND (l.end_date IS NULL OR l.end_date >= current_month_start)
    LOOP
        -- Calculate due date for current month
        current_due_date := current_month_start + (COALESCE(lease_record.rent_due_day, 1) - 1) * INTERVAL '1 day';
        
        -- Validate the due date doesn't exceed month boundaries
        IF current_due_date >= (current_month_start + INTERVAL '1 month') THEN
            -- If rent_due_day is beyond the days in the month, use last day of month
            current_due_date := (current_month_start + INTERVAL '1 month' - INTERVAL '1 day')::DATE;
        END IF;
        
        -- Only create if this rent payment doesn't already exist
        IF NOT EXISTS (
            SELECT 1 FROM rent_payments 
            WHERE lease_id = lease_record.id 
            AND due_date = current_due_date
        ) THEN
            -- Insert new rent payment record
            INSERT INTO rent_payments (
                lease_id,
                due_date,
                amount_due,
                amount_paid,
                late_fee,
                payment_status,
                created_at
            ) VALUES (
                lease_record.id,
                current_due_date,
                lease_record.monthly_rent,
                0,
                COALESCE(lease_record.late_fee, 0),
                'pending',
                CURRENT_TIMESTAMP
            );
            
            affected_rows := affected_rows + 1;
        END IF;
    END LOOP;
    
    -- Log the execution in cron_job_log
    INSERT INTO cron_job_log (
        job_name, 
        records_affected, 
        success,
        duration_seconds,
        job_parameters
    ) VALUES (
        'generate_monthly_rent_payments', 
        affected_rows, 
        true,
        EXTRACT(EPOCH FROM (CURRENT_TIMESTAMP - start_time)),
        jsonb_build_object(
            'month', EXTRACT(MONTH FROM current_month_start),
            'year', EXTRACT(YEAR FROM current_month_start)
        )
    );
    
    -- Optional: Log in user_activity_log if we have a system user
    IF system_user_id IS NOT NULL THEN
        INSERT INTO user_activity_log (
            user_id,
            activity_type,
            activity_description,
            affected_resource_type
        ) VALUES (
            system_user_id,
            'system_rent_generation',
            'Generated ' || affected_rows || ' rent payment records for ' || TO_CHAR(current_due_date, 'Month YYYY'),
            'rent_payment'
        );
    END IF;
    
EXCEPTION WHEN OTHERS THEN
    -- Log any errors
    INSERT INTO cron_job_log (
        job_name, 
        success, 
        error_message, 
        records_affected,
        duration_seconds
    ) VALUES (
        'generate_monthly_rent_payments', 
        false, 
        SQLERRM, 
        affected_rows,
        EXTRACT(EPOCH FROM (CURRENT_TIMESTAMP - start_time))
    );
    
    RAISE NOTICE 'Error in generate_monthly_rent_payments: %', SQLERRM;
    RAISE;
END;
$$;

COMMENT ON FUNCTION generate_monthly_rent_payments() IS 'Generates rent payment records for active leases';

-- Update overdue rent payments
CREATE OR REPLACE FUNCTION update_overdue_rent_payments()
RETURNS void
LANGUAGE plpgsql
AS $$
DECLARE
    updated_rows INTEGER := 0;
    start_time TIMESTAMP;
BEGIN
    start_time := CURRENT_TIMESTAMP;
    
    -- Update payments that are past due (considering grace period)
    UPDATE rent_payments rp
    SET 
        payment_status = 'overdue',
        updated_at = CURRENT_TIMESTAMP
    FROM leases l
    WHERE rp.lease_id = l.id
    AND rp.payment_status = 'pending'
    AND rp.due_date + COALESCE(l.grace_period_days, 0) < CURRENT_DATE;
    
    GET DIAGNOSTICS updated_rows = ROW_COUNT;
    
    -- Log the execution
    INSERT INTO cron_job_log (
        job_name, 
        records_affected, 
        success,
        duration_seconds
    ) VALUES (
        'update_overdue_rent_payments', 
        updated_rows, 
        true,
        EXTRACT(EPOCH FROM (CURRENT_TIMESTAMP - start_time))
    );
    
EXCEPTION WHEN OTHERS THEN
    INSERT INTO cron_job_log (
        job_name, 
        success, 
        error_message,
        duration_seconds
    ) VALUES (
        'update_overdue_rent_payments', 
        false, 
        SQLERRM,
        EXTRACT(EPOCH FROM (CURRENT_TIMESTAMP - start_time))
    );
    RAISE;
END;
$$;

-- Update lease status by dates
CREATE OR REPLACE FUNCTION update_lease_status_by_dates()
RETURNS void
LANGUAGE plpgsql
AS $$
DECLARE
    activated_count INTEGER := 0;
    expired_count INTEGER := 0;
    terminated_count INTEGER := 0;
    start_time TIMESTAMP;
BEGIN
    start_time := CURRENT_TIMESTAMP;
    
    -- 1. Set lease_status to 'active' where lease has started and is still ongoing
    UPDATE leases
    SET lease_status = 'active',
        updated_at = CURRENT_TIMESTAMP
    WHERE lease_status IN ('draft', 'renewed')
      AND start_date <= CURRENT_DATE
      AND (end_date IS NULL OR end_date >= CURRENT_DATE);
    
    GET DIAGNOSTICS activated_count = ROW_COUNT;

    -- 2. Set lease_status to 'expired' where end_date has passed
    UPDATE leases
    SET lease_status = 'expired',
        updated_at = CURRENT_TIMESTAMP
    WHERE lease_status = 'active'
      AND end_date IS NOT NULL
      AND end_date < CURRENT_DATE;
    
    GET DIAGNOSTICS expired_count = ROW_COUNT;

    -- 3. Set lease_status to 'terminated' where move_out_date has passed
    UPDATE leases
    SET lease_status = 'terminated',
        updated_at = CURRENT_TIMESTAMP
    WHERE lease_status = 'active'
      AND move_out_date IS NOT NULL
      AND move_out_date < CURRENT_DATE;
    
    GET DIAGNOSTICS terminated_count = ROW_COUNT;
    
    -- Log the execution
    INSERT INTO cron_job_log (
        job_name, 
        records_affected, 
        success,
        duration_seconds,
        execution_details
    ) VALUES (
        'update_lease_status_by_dates', 
        activated_count + expired_count + terminated_count, 
        true,
        EXTRACT(EPOCH FROM (CURRENT_TIMESTAMP - start_time)),
        jsonb_build_object(
            'activated', activated_count,
            'expired', expired_count,
            'terminated', terminated_count
        )
    );
    
EXCEPTION WHEN OTHERS THEN
    INSERT INTO cron_job_log (
        job_name, 
        success, 
        error_message,
        duration_seconds
    ) VALUES (
        'update_lease_status_by_dates', 
        false, 
        SQLERRM,
        EXTRACT(EPOCH FROM (CURRENT_TIMESTAMP - start_time))
    );
    RAISE;
END;
$$;

COMMENT ON FUNCTION update_lease_status_by_dates() IS 'Updates lease statuses based on dates';

-- Helper function to generate rent payments for a specific month
CREATE OR REPLACE FUNCTION generate_rent_payments_for_month(
    target_year INTEGER, 
    target_month INTEGER
)
RETURNS TABLE(
    lease_id INTEGER,
    lease_number VARCHAR(100),
    created_payment_id INTEGER,
    due_date DATE,
    amount_due NUMERIC(10,2)
)
LANGUAGE plpgsql
AS $$
DECLARE
    lease_record RECORD;
    month_start DATE;
    payment_due_date DATE;
    new_payment_id INTEGER;
    payments_created INTEGER := 0;
    start_time TIMESTAMP;
BEGIN
    start_time := CURRENT_TIMESTAMP;
    
    -- Validate input parameters
    IF target_month < 1 OR target_month > 12 THEN
        RAISE EXCEPTION 'Invalid month: %. Must be between 1 and 12.', target_month;
    END IF;
    
    IF target_year < 2020 OR target_year > 2100 THEN
        RAISE EXCEPTION 'Invalid year: %. Must be between 2020 and 2100.', target_year;
    END IF;
    
    -- Calculate the target month start
    month_start := DATE(target_year || '-' || LPAD(target_month::TEXT, 2, '0') || '-01');
    
    -- Loop through all active leases
    FOR lease_record IN
        SELECT 
            l.id,
            l.lease_number,
            l.monthly_rent,
            l.rent_due_day,
            l.late_fee,
            l.grace_period_days
        FROM leases l
        WHERE l.lease_status = 'active'
        AND l.start_date <= month_start + INTERVAL '1 month' - INTERVAL '1 day'
        AND (l.end_date IS NULL OR l.end_date >= month_start)
        AND NOT EXISTS (
            SELECT 1 FROM rent_payments rp 
            WHERE rp.lease_id = l.id 
            AND EXTRACT(MONTH FROM rp.due_date) = target_month 
            AND EXTRACT(YEAR FROM rp.due_date) = target_year
        )
    LOOP
        -- Calculate due date for this lease
        payment_due_date := month_start + (COALESCE(lease_record.rent_due_day, 1) - 1) * INTERVAL '1 day';
        
        -- Ensure due date doesn't exceed month boundaries
        IF payment_due_date >= (month_start + INTERVAL '1 month') THEN
            payment_due_date := (month_start + INTERVAL '1 month' - INTERVAL '1 day')::DATE;
        END IF;
        
        -- Create rent payment record
        INSERT INTO rent_payments (
            lease_id,
            due_date,
            amount_due,
            amount_paid,
            late_fee,
            payment_status
        ) VALUES (
            lease_record.id,
            payment_due_date,
            lease_record.monthly_rent,
            0,
            COALESCE(lease_record.late_fee, 0),
            CASE 
                WHEN payment_due_date + COALESCE(lease_record.grace_period_days, 0) < CURRENT_DATE THEN 'overdue'
                WHEN payment_due_date < CURRENT_DATE THEN 'pending'
                ELSE 'pending'
            END
        ) RETURNING id INTO new_payment_id;
        
        payments_created := payments_created + 1;
        
        -- Return the created payment info
        RETURN QUERY SELECT 
            lease_record.id,
            lease_record.lease_number,
            new_payment_id,
            payment_due_date,
            lease_record.monthly_rent;
    END LOOP;
    
    -- Log the execution
    INSERT INTO cron_job_log (
        job_name, 
        records_affected, 
        success,
        duration_seconds,
        job_parameters
    ) VALUES (
        'generate_rent_payments_for_month',
        payments_created,
        true,
        EXTRACT(EPOCH FROM (CURRENT_TIMESTAMP - start_time)),
        jsonb_build_object(
            'year', target_year,
            'month', target_month
        )
    );
END;
$$;

COMMENT ON FUNCTION generate_rent_payments_for_month(INTEGER, INTEGER) IS 'Generates rent payments for a specific month';

-- Calculate monthly expenses for a period
CREATE OR REPLACE FUNCTION calculate_monthly_expenses(
    start_period DATE, 
    end_period DATE, 
    target_property_id INTEGER DEFAULT NULL
)
RETURNS TABLE(
    property_id INTEGER, 
    expense_type VARCHAR, 
    total_amount NUMERIC,
    calculation_details TEXT
) 
LANGUAGE plpgsql
AS $$
DECLARE
    period_months NUMERIC;
    period_days INTEGER;
BEGIN
    -- Validate input parameters
    IF start_period > end_period THEN
        RAISE EXCEPTION 'Start period (%) cannot be after end period (%)', start_period, end_period;
    END IF;
    
    -- Calculate the period length in months and days
    period_months := EXTRACT(YEAR FROM AGE(end_period, start_period)) * 12 + 
                    EXTRACT(MONTH FROM AGE(end_period, start_period)) +
                    EXTRACT(DAY FROM AGE(end_period, start_period)) / 30.44;
    
    period_days := end_period - start_period + 1;
    
    RETURN QUERY
    SELECT 
        pe.property_id,
        pe.expense_type,
        CASE 
            WHEN pe.frequency = 'monthly' THEN 
                ROUND(pe.amount * period_months, 2)
            
            WHEN pe.frequency = 'quarterly' THEN 
                ROUND(pe.amount * (period_months / 3.0), 2)
            
            WHEN pe.frequency = 'semi-annual' THEN 
                ROUND(pe.amount * (period_months / 6.0), 2)
            
            WHEN pe.frequency = 'annual' THEN 
                ROUND(pe.amount * (period_months / 12.0), 2)
            
            WHEN pe.frequency = 'one-time' THEN
                CASE 
                    WHEN pe.expense_date BETWEEN start_period AND end_period THEN pe.amount
                    ELSE 0::NUMERIC
                END
            
            ELSE 0::NUMERIC
        END AS total_amount,
        
        -- Add calculation details for transparency
        CASE 
            WHEN pe.frequency = 'monthly' THEN 
                'Monthly: $' || pe.amount || ' × ' || ROUND(period_months, 2) || ' months'
            
            WHEN pe.frequency = 'quarterly' THEN 
                'Quarterly: $' || pe.amount || ' × ' || ROUND(period_months / 3.0, 2) || ' quarters'
            
            WHEN pe.frequency = 'semi-annual' THEN 
                'Semi-annual: $' || pe.amount || ' × ' || ROUND(period_months / 6.0, 2) || ' periods'
            
            WHEN pe.frequency = 'annual' THEN 
                'Annual: $' || pe.amount || ' × ' || ROUND(period_months / 12.0, 2) || ' years'
            
            WHEN pe.frequency = 'one-time' THEN
                CASE 
                    WHEN pe.expense_date BETWEEN start_period AND end_period THEN 
                        'One-time expense on ' || pe.expense_date::TEXT
                    ELSE 'One-time expense outside period (excluded)'
                END
            
            ELSE 'Unknown frequency'
        END AS calculation_details
        
    FROM property_expenses pe
    WHERE pe.is_active = true
    AND (target_property_id IS NULL OR pe.property_id = target_property_id)
    AND (pe.end_date IS NULL OR pe.end_date >= start_period)
    AND pe.start_date <= end_period
    AND pe.amount > 0
    ORDER BY pe.property_id, pe.expense_type;
END;
$$;

COMMENT ON FUNCTION calculate_monthly_expenses(DATE, DATE, INTEGER) IS 'Calculates prorated expenses for a period';

-- Property expense summary
CREATE OR REPLACE FUNCTION calculate_property_expense_summary(
    start_period DATE, 
    end_period DATE, 
    target_property_id INTEGER DEFAULT NULL
)
RETURNS TABLE(
    property_id INTEGER,
    property_name VARCHAR,
    total_monthly_expenses NUMERIC,
    total_quarterly_expenses NUMERIC,
    total_annual_expenses NUMERIC,
    total_onetime_expenses NUMERIC,
    grand_total NUMERIC,
    expense_count BIGINT
) 
LANGUAGE plpgsql
AS $$
BEGIN
    RETURN QUERY
    WITH expense_calculations AS (
        SELECT * FROM calculate_monthly_expenses(start_period, end_period, target_property_id)
    )
    SELECT 
        p.id AS property_id,
        p.property_name,
        COALESCE(SUM(CASE WHEN pe.frequency = 'monthly' THEN ec.total_amount ELSE 0 END), 0) AS total_monthly_expenses,
        COALESCE(SUM(CASE WHEN pe.frequency = 'quarterly' THEN ec.total_amount ELSE 0 END), 0) AS total_quarterly_expenses,
        COALESCE(SUM(CASE WHEN pe.frequency = 'annual' THEN ec.total_amount ELSE 0 END), 0) AS total_annual_expenses,
        COALESCE(SUM(CASE WHEN pe.frequency = 'one-time' THEN ec.total_amount ELSE 0 END), 0) AS total_onetime_expenses,
        COALESCE(SUM(ec.total_amount), 0) AS grand_total,
        COUNT(ec.*) AS expense_count
    FROM properties p
    LEFT JOIN property_expenses pe ON p.id = pe.property_id
    LEFT JOIN expense_calculations ec ON p.id = ec.property_id AND pe.expense_type = ec.expense_type
    WHERE (target_property_id IS NULL OR p.id = target_property_id)
    GROUP BY p.id, p.property_name
    HAVING COUNT(ec.*) > 0 OR target_property_id IS NOT NULL
    ORDER BY p.property_name;
END;
$$;

COMMENT ON FUNCTION calculate_property_expense_summary(DATE, DATE, INTEGER) IS 'Summarizes expenses by property';

-- Expense breakdown by category
CREATE OR REPLACE FUNCTION calculate_expenses_by_category(
    start_period DATE, 
    end_period DATE, 
    target_property_id INTEGER DEFAULT NULL
)
RETURNS TABLE(
    category_name VARCHAR,
    expense_count BIGINT,
    total_amount NUMERIC,
    avg_amount NUMERIC,
    frequency_breakdown JSONB
) 
LANGUAGE plpgsql
AS $$
BEGIN
    RETURN QUERY
    WITH expense_data AS (
        SELECT 
            COALESCE(ec.category_name, 'Uncategorized') AS category,
            pe.frequency,
            ce.total_amount
        FROM calculate_monthly_expenses(start_period, end_period, target_property_id) ce
        JOIN property_expenses pe ON ce.property_id = pe.property_id AND ce.expense_type = pe.expense_type
        LEFT JOIN expense_categories ec ON pe.category_id = ec.id
        WHERE ce.total_amount > 0
    )
    SELECT 
        ed.category AS category_name,
        COUNT(*)::BIGINT AS expense_count,
        ROUND(SUM(ed.total_amount), 2) AS total_amount,
        ROUND(AVG(ed.total_amount), 2) AS avg_amount,
        jsonb_object_agg(
            ed.frequency, 
            jsonb_build_object(
                'count', COUNT(*),
                'total', ROUND(SUM(ed.total_amount), 2)
            )
        ) AS frequency_breakdown
    FROM expense_data ed
    GROUP BY ed.category
    ORDER BY SUM(ed.total_amount) DESC;
END;
$$;

COMMENT ON FUNCTION calculate_expenses_by_category(DATE, DATE, INTEGER) IS 'Groups expenses by category';

-- Active leases with tenant information
CREATE OR REPLACE VIEW active_leases_with_tenants AS
SELECT 
    l.id AS lease_id,
    l.lease_number,
    l.start_date,
    l.end_date,
    l.monthly_rent,
    l.lease_status,
    p.id AS property_id,
    p.property_name,
    u.id AS unit_id,
    u.unit_number,
    STRING_AGG(
        DISTINCT (t.first_name || ' ' || t.last_name), 
        ', ' 
        ORDER BY t.first_name || ' ' || t.last_name
    ) FILTER (WHERE lt.is_primary_tenant = true) AS primary_tenant_names,
    STRING_AGG(
        DISTINCT (t.first_name || ' ' || t.last_name), 
        ', ' 
        ORDER BY t.first_name || ' ' || t.last_name
    ) AS all_tenant_names,
    STRING_AGG(DISTINCT t.email, ', ' ORDER BY t.email) AS tenant_emails,
    STRING_AGG(DISTINCT t.phone, ', ' ORDER BY t.phone) FILTER (WHERE t.phone IS NOT NULL) AS tenant_phones,
    COUNT(DISTINCT lt.tenant_id) AS tenant_count
FROM leases l
JOIN units u ON l.unit_id = u.id
JOIN properties p ON u.property_id = p.id
LEFT JOIN lease_tenants lt ON l.id = lt.lease_id AND lt.removed_date IS NULL
LEFT JOIN tenants t ON lt.tenant_id = t.id
WHERE l.lease_status = 'active'
GROUP BY l.id, l.lease_number, l.start_date, l.end_date, l.monthly_rent, l.lease_status, 
         p.id, p.property_name, u.id, u.unit_number;

COMMENT ON VIEW active_leases_with_tenants IS 'Active leases with aggregated tenant information';

-- Property summary with occupancy
CREATE OR REPLACE VIEW property_summary AS
SELECT 
    p.id,
    p.property_name,
    p.address,
    p.property_type,
    p.total_units,
    p.is_active,
    COALESCE(COUNT(u.id), 0) AS actual_units,
    COALESCE(SUM(CASE WHEN u.occupancy_status = 'occupied' THEN 1 ELSE 0 END), 0) AS occupied_units,
    COALESCE(SUM(CASE WHEN u.occupancy_status = 'vacant' THEN 1 ELSE 0 END), 0) AS vacant_units,
    COALESCE(SUM(CASE WHEN u.occupancy_status = 'maintenance' THEN 1 ELSE 0 END), 0) AS maintenance_units,
    CASE 
        WHEN COUNT(u.id) > 0 THEN 
            ROUND(SUM(CASE WHEN u.occupancy_status = 'occupied' THEN 1 ELSE 0 END)::NUMERIC / COUNT(u.id) * 100, 2)
        ELSE 0
    END AS occupancy_percentage,
    COALESCE(SUM(u.monthly_rent), 0) AS total_potential_rent,
    COALESCE(
        SUM(CASE WHEN u.occupancy_status = 'occupied' THEN u.monthly_rent ELSE 0 END), 
        0
    ) AS total_actual_rent
FROM properties p
LEFT JOIN units u ON p.id = u.property_id
GROUP BY p.id, p.property_name, p.address, p.property_type, p.total_units, p.is_active;

COMMENT ON VIEW property_summary IS 'Property occupancy and revenue summary';

        JOIN tenants t ON lt.tenant_id = t.id
        WHERE lt.lease_id = l.id 
        AND lt.is_primary_tenant = true 
        AND lt.removed_date IS NULL
        LIMIT 1
    ) AS primary_tenant_name,
    (
        SELECT t.email
        FROM lease_tenants lt
        JOIN tenants t ON lt.tenant_id = t.id
        WHERE lt.lease_id = l.id 
        AND lt.is_primary_tenant = true 
        AND lt.removed_date IS NULL
        LIMIT 1
    ) AS primary_tenant_email,
    CASE 
        WHEN rp.payment_status = 'overdue' THEN 
            GREATEST(0, CURRENT_DATE - rp.due_date - COALESCE(l.grace_period_days, 0))
        ELSE 0
    END AS days_overdue,
    rp.amount_due - rp.amount_paid AS balance_due,
    rp.amount_due + COALESCE(rp.late_fee, 0) AS total_amount_due
FROM rent_payments rp
JOIN leases l ON rp.lease_id = l.id
JOIN units u ON l.unit_id = u.id
JOIN properties p ON u.property_id = p.id;

COMMENT ON VIEW rent_collection_view IS 'Comprehensive rent payment tracking';

-- Pending payment submissions
CREATE OR REPLACE VIEW pending_payment_submissions AS

    t.id AS tenant_id,
    (t.first_name || ' ' || t.last_name) AS tenant_name,

FROM payment_submissions ps

JOIN units u ON l.unit_id = u.id
JOIN properties p ON u.property_id = p.id
WHERE ps.verification_status = 'pending'
ORDER BY ps.submission_date;

COMMENT ON VIEW pending_payment_submissions IS 'Payment submissions awaiting verification';

-- Blacklisted tenants summary
CREATE OR REPLACE VIEW blacklisted_tenants_summary AS
SELECT 
    t.id,
    (t.first_name || ' ' || t.last_name) AS full_name,
    t.email,
    t.phone,
    t.blacklist_reason,
    t.blacklist_severity,
    t.blacklisted_date,
    t.blacklisted_by,
    bc.category_name AS blacklist_category,
    CASE 
        WHEN p.property_name IS NOT NULL THEN (p.property_name || ' - Unit ' || u.unit_number)
        ELSE 'No Current Lease'
    END AS current_property,
    l.lease_number AS current_lease_number,
    (SELECT COUNT(*) FROM tenant_blacklist_history bh WHERE bh.tenant_id = t.id) AS history_entries,
    EXTRACT(DAY FROM CURRENT_TIMESTAMP - t.blacklisted_date)::INTEGER AS days_blacklisted,
    t.blacklist_expires_at,
    CASE 
        WHEN t.blacklist_expires_at IS NOT NULL AND t.blacklist_expires_at < CURRENT_DATE THEN true
        ELSE false
    END AS is_expired
FROM tenants t
LEFT JOIN blacklist_categories bc ON t.blacklist_category_id = bc.id
LEFT JOIN lease_tenants lt ON t.id = lt.tenant_id AND lt.removed_date IS NULL
LEFT JOIN leases l ON lt.lease_id = l.id AND l.lease_status = 'active'
LEFT JOIN units u ON l.unit_id = u.id
LEFT JOIN properties p ON u.property_id = p.id
WHERE t.is_blacklisted = true
ORDER BY t.blacklisted_date DESC;

COMMENT ON VIEW blacklisted_tenants_summary IS 'Summary of blacklisted tenants';

-- Open maintenance requests
CREATE OR REPLACE VIEW open_maintenance_requests AS
SELECT 
    mr.id,
    mr.request_title,
    mr.description,
    mr.priority,
    mr.category,
    mr.status,
    mr.requested_date,
    mr.scheduled_date,
    mr.estimated_cost,
    p.id AS property_id,
    p.property_name,
    u.id AS unit_id,
    u.unit_number,
    t.id AS tenant_id,
    CONCAT(t.first_name, ' ', t.last_name) AS tenant_name,
    t.phone AS tenant_phone,
    CONCAT(u_assigned.first_name, ' ', u_assigned.last_name) AS assigned_to_name,
    EXTRACT(DAY FROM CURRENT_TIMESTAMP - mr.requested_date)::INTEGER AS days_open
FROM maintenance_requests mr
JOIN units u ON mr.unit_id = u.id
JOIN properties p ON u.property_id = p.id
LEFT JOIN tenants t ON mr.tenant_id = t.id
LEFT JOIN users u_assigned ON mr.assigned_to = u_assigned.id
WHERE mr.status IN ('open', 'assigned', 'in_progress')
ORDER BY 
    CASE mr.priority
        WHEN 'emergency' THEN 1
        WHEN 'high' THEN 2
        WHEN 'medium' THEN 3
        WHEN 'low' THEN 4
    END,
    mr.requested_date;

COMMENT ON VIEW open_maintenance_requests IS 'Active maintenance requests prioritized';

-- Expiring leases (within 60 days)
CREATE OR REPLACE VIEW expiring_leases AS
SELECT 
    l.id AS lease_id,
    l.lease_number,
    l.start_date,
    l.end_date,
    l.monthly_rent,
    l.auto_renew,
    l.renewal_notice_days,
    p.id AS property_id,
    p.property_name,
    u.id AS unit_id,
    u.unit_number,
    STRING_AGG(
        CONCAT(t.first_name, ' ', t.last_name),
        ', '
        ORDER BY lt.is_primary_tenant DESC
    ) AS tenant_names,
    STRING_AGG(t.email, ', ' ORDER BY lt.is_primary_tenant DESC) AS tenant_emails,
    l.end_date - CURRENT_DATE AS days_until_expiration
FROM leases l
JOIN units u ON l.unit_id = u.id
JOIN properties p ON u.property_id = p.id
JOIN lease_tenants lt ON l.id = lt.lease_id AND lt.removed_date IS NULL
JOIN tenants t ON lt.tenant_id = t.id
WHERE l.lease_status = 'active'
AND l.end_date IS NOT NULL
AND l.end_date BETWEEN CURRENT_DATE AND CURRENT_DATE + INTERVAL '60 days'
GROUP BY l.id, l.lease_number, l.start_date, l.end_date, l.monthly_rent, 
         l.auto_renew, l.renewal_notice_days, p.id, p.property_name, u.id, u.unit_number
ORDER BY l.end_date;

COMMENT ON VIEW expiring_leases IS 'Leases expiring within 60 days';
