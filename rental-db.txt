-- =====================================
-- Lease number generator main function
-- =====================================
CREATE OR REPLACE FUNCTION generate_lease_number(unit_id INT)
RETURNS VARCHAR AS $$
DECLARE
    prop_id INT;
    prop_code VARCHAR(3);
    next_num INT;
BEGIN
    -- Get property info from the unit
    SELECT p.id, LEFT(p.property_name, 3)
    INTO prop_id, prop_code
    FROM properties p
    JOIN units u ON p.id = u.property_id
    WHERE u.id = unit_id;

    IF prop_id IS NULL THEN
        RAISE EXCEPTION 'Unit % not linked to any property', unit_id;
    END IF;

    -- Get next available number for this unit
    SELECT COALESCE(MAX(CAST(RIGHT(l.lease_number, 3) AS INTEGER)), 0) + 1
    INTO next_num
    FROM leases l
    WHERE l.unit_id = unit_id;

    -- Return formatted lease number e.g. APA-001
    RETURN UPPER(prop_code) || '-' || TO_CHAR(next_num, 'FM000');
END;
$$ LANGUAGE plpgsql;


-- =====================================
-- Trigger wrapper function
-- =====================================
CREATE OR REPLACE FUNCTION set_lease_number()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.lease_number IS NULL THEN
        NEW.lease_number := generate_lease_number(NEW.unit_id);
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- =====================================
-- Attach trigger to leases
-- =====================================
DROP TRIGGER IF EXISTS trigger_generate_lease_number ON leases CASCADE;
CREATE TRIGGER trigger_generate_lease_number
BEFORE INSERT ON leases
FOR EACH ROW
EXECUTE FUNCTION set_lease_number();

-- ===============================
-- Function: calculate monthly expenses
-- ===============================
CREATE OR REPLACE FUNCTION calculate_monthly_expenses(p_property_id INT)
RETURNS DECIMAL(10,2) AS $$
DECLARE
    total DECIMAL(10,2);
BEGIN
    SELECT COALESCE(SUM(
        CASE 
            WHEN frequency = 'monthly' THEN amount
            WHEN frequency = 'quarterly' THEN amount / 3
            WHEN frequency = 'annual' THEN amount / 12
            ELSE amount
        END
    ), 0)
    INTO total
    FROM property_expenses
    WHERE property_id = p_property_id
      AND is_active = true;

    RETURN total;
END;
$$ LANGUAGE plpgsql;

-- Function: process verified payment submissions
-- ===============================
CREATE OR REPLACE FUNCTION process_verified_payment_submission(p_submission_id INT)
RETURNS VOID AS $$
DECLARE
    sub_rec RECORD;
    rent_rec RECORD;
    new_total DECIMAL(10,2);
BEGIN
    -- Get submission
    SELECT * INTO sub_rec
    FROM payment_submissions
    WHERE id = p_submission_id;

    -- Only process if verified
    IF sub_rec.verification_status = 'verified' THEN
        -- Find first pending rent payment
        SELECT * INTO rent_rec
        FROM rent_payments
        WHERE lease_id = sub_rec.lease_id
          AND payment_status = 'pending'
        ORDER BY due_date ASC
        LIMIT 1;

        IF FOUND THEN
            new_total := rent_rec.amount_paid + sub_rec.amount;

            UPDATE rent_payments
            SET amount_paid = new_total,
                payment_date = sub_rec.transaction_date,
                payment_method = INITCAP(sub_rec.payment_method),
                payment_reference = sub_rec.transaction_reference,
                payment_status = CASE 
                                    WHEN new_total >= rent_rec.amount_due THEN 'paid'
                                    ELSE 'partial'
                                 END,
                updated_at = CURRENT_TIMESTAMP
            WHERE id = rent_rec.id;

            PERFORM log_user_activity(sub_rec.verified_by, 'verify_payment',
                'payment_submission', sub_rec.id,
                'Processed verified payment submission');
        END IF;
    END IF;
END;
$$ LANGUAGE plpgsql;


-- ===============================
-- Validation trigger function for users
-- ===============================
CREATE OR REPLACE FUNCTION validate_user_tenant_relationship()
RETURNS TRIGGER AS $$
DECLARE
    tenant_role_id INT;
BEGIN
    SELECT id INTO tenant_role_id FROM user_roles WHERE role_name = 'Tenant';
    IF tenant_role_id IS NULL THEN
        RAISE EXCEPTION 'Tenant role not seeded in user_roles';
    END IF;

    IF NEW.role_id = tenant_role_id AND NEW.tenant_id IS NULL THEN
        RAISE EXCEPTION 'Tenant users must be linked to a tenant record';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS validate_user_tenant_relationship_trigger ON users CASCADE;
CREATE TRIGGER validate_user_tenant_relationship_trigger
BEFORE INSERT OR UPDATE ON users
FOR EACH ROW
EXECUTE FUNCTION validate_user_tenant_relationship();

-- ===============================
-- Views
-- ===============================
DROP VIEW IF EXISTS lease_summary CASCADE;
CREATE OR REPLACE VIEW lease_summary AS
SELECT 
    l.id AS lease_id,
    l.lease_number,
    p.property_name,
    u.unit_number,
    l.start_date,
    l.end_date,
    l.monthly_rent,
    STRING_AGG(t.first_name || ' ' || t.last_name, ', ' ORDER BY lt.is_primary_tenant DESC) AS tenants,
    l.lease_status
FROM leases l
JOIN units u ON l.unit_id = u.id
JOIN properties p ON u.property_id = p.id
JOIN lease_tenants lt ON l.id = lt.lease_id
JOIN tenants t ON lt.tenant_id = t.id
GROUP BY l.id, p.property_name, u.unit_number;

DROP VIEW IF EXISTS tenant_portal_info CASCADE;
CREATE OR REPLACE VIEW tenant_portal_info AS
SELECT 
    t.id AS tenant_id,
    t.first_name,
    t.last_name,
    t.email,
    u.username,
    u.is_active,
    l.lease_number,
    l.lease_status,
    l.start_date,
    l.end_date,
    l.monthly_rent,
    p.property_name,
    un.unit_number
FROM tenants t
JOIN users u ON t.id = u.tenant_id
JOIN user_roles ur ON u.role_id = ur.id
LEFT JOIN lease_tenants lt ON t.id = lt.tenant_id
LEFT JOIN leases l ON lt.lease_id = l.id
LEFT JOIN units un ON l.unit_id = un.id
LEFT JOIN properties p ON un.property_id = p.id
WHERE ur.role_name = 'Tenant';

COMMENT ON TABLE rent_payments IS 'Tracks rent payments for leases';

COMMENT ON TABLE tenants IS 'Tenant personal information and rental history';
COMMENT ON TABLE properties IS 'Properties managed in the system';
COMMENT ON TABLE units IS 'Individual units within properties';
